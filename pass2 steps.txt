import java.io.*;
import java.util.*;

class Pass2 {
    public static void main(String[] args) throws Exception {
        BufferedReader inter = new BufferedReader(new FileReader("intermediate.txt"));
        BufferedReader symtab = new BufferedReader(new FileReader("symtab.txt"));
        BufferedReader optab = new BufferedReader(new FileReader("optab.txt"));

        Map<String, String> sym = new HashMap<>();
        Map<String, String> op = new HashMap<>();

        // Load symbol table
        String line;
        while ((line = symtab.readLine()) != null) {
            String[] parts = line.trim().split("\\s+");
            sym.put(parts[0], parts[1]);
        }

        // Load opcode table
        while ((line = optab.readLine()) != null) {
            String[] parts = line.trim().split("\\s+");
            op.put(parts[0], parts[1]);
        }

        BufferedWriter machine = new BufferedWriter(new FileWriter("machine.txt"));

        // Skip START
        line = inter.readLine(); // skip START line

        while ((line = inter.readLine()) != null) {
            String[] parts = line.trim().split("\\s+");

            if (parts.length < 4) continue;

            String loc = parts[0];
            String label = parts[1];
            String opcode = parts[2];
            String operand = parts[3];

            String machineCode = loc + "\t";

            if (op.containsKey(opcode)) {
                machineCode += op.get(opcode) + "\t";
                String[] reg_op = operand.split(",");

                // Handle immediate values
                if (reg_op[1].contains("=")) {
                    machineCode += reg_op[0] + "\t" + reg_op[1].replace("='", "").replace("'", "");
                } else {
                    machineCode += reg_op[0] + "\t" + sym.getOrDefault(reg_op[1], "000");
                }
            } else {
                machineCode += "-\t-\t-"; // For non-instruction lines like DS, END
            }

            machine.write(machineCode + "\n");
        }

        inter.close();
        symtab.close();
        optab.close();
        machine.close();

        System.out.println("Pass 2 completed. Check machine.txt for output.");
    }
}
