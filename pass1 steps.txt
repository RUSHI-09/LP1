import java.io.*;
import java.util.*;

class Pass1 {
    public static void main(String[] args) throws Exception {
        BufferedReader br = new BufferedReader(new FileReader("input.asm"));
        BufferedWriter symtab = new BufferedWriter(new FileWriter("symtab.txt"));
        BufferedWriter intermediate = new BufferedWriter(new FileWriter("intermediate.txt"));

        String line = br.readLine();
        String[] parts = line.split(" ");
        int locctr = Integer.parseInt(parts[1]);

        intermediate.write(line + "\n");

        while ((line = br.readLine()) != null) {
            parts = line.split(" ");
            String label = parts[0];
            String opcode = parts[1];

            if (!label.equals("**")) {
                symtab.write(label + "\t" + locctr + "\n");
            }

            intermediate.write(locctr + "\t" + line + "\n");
            locctr += 1;
        }

        br.close();
        symtab.close();
        intermediate.close();

        System.out.println("Pass 1 completed. Check symtab.txt and intermediate.txt.");
    }
}
